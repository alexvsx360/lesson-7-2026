Scope — איפה המשתנה «חי»?

Scope = התחום שבו משתנה או פונקציה נגישים. מחוץ לתחום — JavaScript לא מכיר את השם.

const appName = "Course App";
let counter = 0;

function increment() {
    counter++;
    console.log(`${appName} — counter:`, counter);
}

increment();
increment();
כלל אצבע: הגדירו משתנים במקום הכי צר שבו צריכים אותם — לא גלובלי «סתם».

Scope Chain ו-Shadowing

Scope Chain — JS מחפש משתנה מבפנים החוצה. Shadowing — משתנה פנימי עם אותו שם «מסתיר» את החיצוני.

const city = "תל אביב";
function outer() {
    const city = "חיפה";
    function inner() {
        const city = "ירושלים";
        console.log("inner:", city);
    }
    inner();
    console.log("outer:", city);
}
outer();
console.log("גלובלי:", city);

Scope בלולאות

משתנה של let בתוך for קיים רק בתוך הלולאה — זה עוזר למנוע טעויות.

const results = [];
for (let i = 0; i < 3; i++) {
    const label = `פריט ${i}`;
    results.push(label);
}
console.log("תוצאות:", results);
// console.log(i); // שגיאה עם let

Validation — אימות קלט

Validation = לוודא שהנתונים שהמשתמש שולח תקינים לפני שממשיכים (שמירה, שליחה, חישוב).

בלי Validation — טופס ריק, גיל שלילי, אימייל בלי @ — עלולים לשבור את האפליקציה או לשמור מידע שגוי.

  • האם השדה ריק?
  • האם זה מספר / טקסט?
  • האם הערך בטווח הגיוני? (גיל 18+, אורך סיסמה…)

Validation בסיסי

const name = "   ";
const age = "17";

console.log("שם ריק?", name.trim() === "");
console.log("גיל מספר?", !Number.isNaN(Number(age)));
console.log("גיל תקין (18+)?", Number(age) >= 18);
trim()מסיר רווחים מהתחלה וסוף — « » נחשב ריק.
typeofבודק סוג — מספר, מחרוזת, בוליאני.
תנאיםמשלבים בדיקות עם && ו-||.

Validation — אימייל ותנאים

function isValidEmail(email) {
    return email.includes("@") && email.includes(".") && email.length >= 5;
}
console.log("test@mail.com:", isValidEmail("test@mail.com"));
console.log("bad-email:", isValidEmail("bad-email"));

בפרויקטים אמיתיים משתמשים בביטויים רגולריים (RegExp) — כאן מספיק בדיקה פשוטה ללמידה.

פונקציית validate מלאה

function validateUser(user) {
    const errors = [];
    if (!user.name || user.name.trim() === "") errors.push("שם חסר");
    if (typeof user.age !== "number" || user.age < 18) errors.push("גיל לא תקין");
    if (!user.email || !user.email.includes("@")) errors.push("אימייל לא תקין");
    return { valid: errors.length === 0, errors };
}
דפוס מומלץ: מחזירים { valid, errors } — קל להציג הודעות למשתמש.

Loops — שלושת הסוגים העיקריים

לולאה אומרת למחשב: «תחזור על אותה פעולה». ב-JavaScript יש שלוש דרכים עיקריות — ההבדל הוא מתי בודקים את התנאי וכמה פעמים בערך נדע לחזור.

for — הכי נפוצה מבנה: for (התחלה; תנאי; צעד). מושלמת כשיודעים כמה פעמים לרוץ — למשל «עבור על כל איבר במערך».
while — כל עוד… בודקת תנאי לפני כל סיבוב. אם התנאי false מההתחלה — לא רצה אפילו פעם. טובה כשלא יודעים מראש כמה פעמים (למשל «עד שהמשתמש יזין נכון»).
do...while — לפחות פעם אחת מריצה את הגוף, ואז בודקת. תמיד רצה מינימום פעם אחת. לדוגמה: תפריט שמציג אפשרויות ושואל «לנסות שוב?». 
// for — יודעים מראש: 1, 2, 3
for (let i = 1; i <= 3; i++) {
    console.log("for:", i);
}

// while — בודק לפני כל סיבוב
let n = 1;
while (n <= 3) {
    console.log("while:", n);
    n++;
}

// do...while — רץ פעם אחת, ואז בודק
let m = 0;
do {
    m++;
    console.log("do...while:", m);
} while (m < 3);
לולאהמתי בודקים?דוגמה מהחיים
forלפני כל סיבובלעבור על 10 פריטים במערך
whileלפני כל סיבובלחכות עד שיש קלט תקין
do...whileאחרי כל סיבובלהציג תפריט לפחות פעם אחת

טיפ: על מערכים מודרניים לעיתים נוח יותר for...of או forEach — נראה בהמשך. for עדיין חשוב כשצריך אינדקס (מיקום במערך).

break ו-continue

for (let i = 1; i <= 8; i++) {
    if (i % 2 !== 0) continue; // מדלג על אי-זוגיים
    if (i > 6) break;          // עוצר אחרי 6
    console.log("זוגי:", i);
}

slice מול splice

שתי המילים נשמעות דומה — אבל ההבדל קריטי: slice מצלם חלק מהמערך (בלי לגעת במקור), ו-splice משנה את המערך המקורי (מוחק ו/או מוסיף באמצע).

אינדקסים לדוגמה: [10, 20, 30, 40, 50] → אינדקס 0 = 10, אינדקס 1 = 20, אינדקס 2 = 30…

slice(start, end) — «חיתוך לצפייה»

  • start — מאיזה אינדקס להתחיל (כולל)
  • end — עד לאיזה אינדקס (לא כולל את end עצמו!)
  • מחזיר מערך חדש — המקור נשאר זהה

דוגמה: slice(1, 4) על [10,20,30,40,50] → [20, 30, 40]

splice(start, deleteCount, ...items) — «עריכה אמיתית»

  • start — מאיזה אינדקס להתחיל
  • deleteCount — כמה איברים למחוק מכאן
  • ...items — (אופציונלי) מה להוסיף במקום
  • מחזיר מערך של מה שנמחק — ומשנה את המערך המקורי

דוגמה: splice(2, 2, 99, 100) — ממיקום 2 מוחק 2 איברים (30, 40) ומכניס 99, 100

מתודהמשנה מקור?מחזירמתי להשתמש?
sliceלא ❌מערך חדש (חתוך)העתקה, הצגה, שליחה בלי לשבור מקור
spliceכן ✅מה שנמחקמחיקה/הוספה באמצע רשימה חיה
 
const nums = [10, 20, 30, 40, 50];

// slice — מקור לא משתנה
const piece = nums.slice(1, 4);
console.log("slice:", piece);   // [20, 30, 40]
console.log("מקור:", nums);     // [10, 20, 30, 40, 50]

// splice — מקור כן משתנה
const removed = nums.splice(2, 2, 99, 100);
console.log("נמחק:", removed);  // [30, 40]
console.log("מקור:", nums);     // [10, 20, 99, 100, 50]
זכרו: slice = צילום. splice = גזירה + הדבקה. אל תתבלבלו — זו טעות נפוצה בכיתה!

מערכים מתקדם: indexOf · includes · join · reverse

אחרי push/pop — ארבע מתודות שמשתמשים בהן כל יום: לחפש, לבדוק קיום, לחבר למחרוזת, ו-להפוך סדר.

indexOf(ערך) מחזיר את האינדקס הראשון שבו הערך נמצא. אם לא נמצא — מחזיר -1. שימושי כשצריך מיקום, לא רק כן/לא.
includes(ערך) מחזיר true או false — «האם הערך קיים?». יותר קריא מ-indexOf !== -1.
join(מפריד) הופך מערך למחרוזת אחת, עם מפריד ביניהם. ["a","b"].join("-") → "a-b". הפוך מ-split על מחרוזת.
reverse() הופך את סדר האיברים במקום — משנה את המערך המקורי! אם צריך עותק הפוך בלי לשבור מקור: [...arr].reverse(). 
const tags = ["js", "html", "css", "js"];

console.log(tags.indexOf("js"));       // 0 — הופעה ראשונה
console.log(tags.indexOf("react"));    // -1 — לא נמצא
console.log(tags.includes("css"));     // true
console.log(tags.includes("react"));    // false
console.log(tags.join(" | "));         // "js | html | css | js"

const nums = [3, 1, 4];
const reversed = [...nums].reverse();   // עותק הפוך — מקור נשמר
console.log("מקור:", nums);            // [3, 1, 4]
console.log("הפוך:", reversed);        // [4, 1, 3]

indexOf מול includes: צריך מיקום? → indexOf. רק לבדוק «יש או אין»? → includes.

אובייקטים: Object.keys() · values() · entries()

אובייקט הוא זוגות של מפתח → ערך. שלוש המתודות האלה הופכות את האובייקט למערכים — כדי שיהיה קל לעבור בלולאה, לספור, או להציג.

מתודהמה מחזיר?דוגמה ל-{ name: "מאיה", role: "student" }
Object.keys(obj)מערך של שמות המפתחות["name", "role"]
Object.values(obj)מערך של הערכים בלבד["מאיה", "student"]
Object.entries(obj)מערך של זוגות [מפתח, ערך][["name","מאיה"], ["role","student"]]
keysכשצריך רשימת שדות — «מה יש באובייקט?»
valuesכשמעניינים רק הערכים — בלי שמות השדות
entriesהכי שימושי ללולאות — מקבלים גם מפתח וגם ערך ביחד
const user = { name: "מאיה", role: "student", active: true };

console.log("keys:", Object.keys(user));
console.log("values:", Object.values(user));
console.log("entries:", Object.entries(user));

// entries + for...of — הדרך המומלצת לעבור על אובייקט
for (const [key, value] of Object.entries(user)) {
    console.log(`${key} => ${value}`);
}

לזכור: Object.keys() מחזיר מערך — אפשר להשתמש ב-.length כדי לדעת כמה שדות יש באובייקט.

Iteration — for · forEach · for...in · for...of

Iteration = מעבר על רשימה או אובייקט איבר־איבר. יש כמה דרכים — הבחירה תלויה במערך או אובייקט ובהאם צריך אינדקס.

for קלאסי for (let i = 0; i < arr.length; i++) — נותן אינדקס i. טוב כשצריך מיקום, או לולאה מורכבת עם break/continue.
forEach arr.forEach((item, i) => …) — קריא ונקי על מערך. אי אפשר לעצור עם break. מתאים ל«תעשה משהו על כל איבר».
for...of for (const x of arr) — עובר על ערכים (מערך, מחרוזת…). מודרני וקריא. אפשר break. לא על אובייקט רגיל!
for...in for (const key in obj) — עובר על מפתחות של אובייקט. על מערכים — עלול לתת אינדקסים מוזרים; לרוב עדיף for...of למערכים.
מה יש לך?מה צריך?מה לבחור?
מערךאינדקס + ערךfor או forEach
מערךרק ערכיםfor...of
אובייקטמפתח + ערךObject.entries + for...of
אובייקטרק מפתחותfor...in או Object.keys
 
const skills = ["HTML", "CSS", "JS"];

// for — עם אינדקס
for (let i = 0; i < skills.length; i++) {
    console.log(i, skills[i]);
}

// forEach — נקי, בלי break
skills.forEach((skill, i) => console.log("forEach:", i, skill));

// for...of — ערכים במערך
for (const skill of skills) console.log("of:", skill);

const config = { theme: "dark", lang: "he" };

// for...in — מפתחות באובייקט
for (const key in config) console.log("in:", key, config[key]);

// entries + for...of — הכי מסודר לאובייקט
for (const [k, v] of Object.entries(config)) console.log("entry:", k, v);
טעות נפוצה: for...in על מערך — עובר גם על מפתחות מיותרים. על מערכים → for...of או forEach.

מבנים מקוננים — מערך של אובייקטים

המצב הנפוץ ביותר באפליקציות: רשימת משתמשים, מוצרים, ציונים — כל פריט הוא אובייקט.

const students = [
    { name: "נועה", grade: 92 },
    { name: "דניאל", grade: 78 },
    { name: "מאיה", grade: 85 }
];
for (const s of students) {
    if (s.grade >= 80) console.log(s.name, s.grade);
}

מבנים מקוננים — אובייקט של מערכים

const classroom = {
    className: "JS 2026",
    students: ["אלכס", "נועה", "דניאל"],
    grades: [90, 82, 76]
};
classroom.students.forEach((name, i) => {
    console.log(`${name}: ${classroom.grades[i]}`);
});

Spread Operator — ...

מפזר אלמנטים ממערך או מפתחות מאובייקט — שימושי להעתקה ומיזוג בלי לשנות את המקור (במערכים).

const a = [1, 2, 3];
const b = [...a, 4, 5];

const user = { name: "דני", age: 20 };
const profile = { ...user, city: "תל אביב", age: 21 };

...user מעתיק את כל המפתחות — age: 21 אחר כך «דורס» את הגיל הישן.

תרגול מסכם בלייב 🔥

Validation + לולאה + Spread — מסננים משתמשים תקינים ומחזירים רשימה חדשה.

function validateAndFilter(users) {
    const valid = [];
    for (const user of users) {
        if (!user.name?.trim()) continue;
        if (user.age < 18) continue;
        valid.push({ ...user, status: "מאושר" });
    }
    return valid;
}
